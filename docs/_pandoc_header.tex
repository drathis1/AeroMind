% Pandoc header includes for AeroMind final report PDF rendering.

% --- Image sizing: constrain every embedded screenshot/diagram so it fits
% the page width and never overflows the page height.
\usepackage{float}
\usepackage{caption}
\captionsetup{font=small,labelfont=bf}
\setkeys{Gin}{width=0.92\linewidth,keepaspectratio}
\let\OldIncludegraphics\includegraphics
% Tighter max height so multiple figures can share a page in the condensed report.
\renewcommand{\includegraphics}[2][]{%
  \OldIncludegraphics[#1,width=0.88\linewidth,height=0.48\textheight,keepaspectratio]{#2}%
}
\renewcommand{\topfraction}{0.9}
\renewcommand{\bottomfraction}{0.85}
\renewcommand{\textfraction}{0.06}
\renewcommand{\floatpagefraction}{0.7}

% (U+2192, U+2194, U+2190 and similar). Map them to a font that has them
